#!/usr/bin/env bash
set -euo pipefail
EXP=<EXPERIMENT_ROOT>
export CUDA_VISIBLE_DEVICES=${GPU_ID:-2}
export MUJOCO_EGL_DEVICE_ID=${GPU_ID:-2}
export EGL_DEVICE_ID=${GPU_ID:-2}
export MUJOCO_GL=egl PYOPENGL_PLATFORM=egl HYDRA_FULL_ERROR=1
export LD_LIBRARY_PATH=<HOME>/.mujoco/mujoco210/bin:/usr/lib/nvidia:/usr/local/cuda/lib64:${LD_LIBRARY_PATH:-}
export PYTHONPATH=$EXP/repo/RL-100:${PYTHONPATH:-}
export OMP_NUM_THREADS=4 OPENBLAS_NUM_THREADS=1
cd "$EXP/repo/RL-100"
exec <LOCAL_PATH> -u train.py --config-name=rl100_2d_epsilon.yaml \
 task=adroit_door_medium hydra.run.dir="$EXP/runs/rgb_seed${SEED:-100}" \
 training.seed=${SEED:-100} training.device=cuda:0 training.resume=True logging.mode=offline \
 checkpoint.save_ckpt=True training.checkpoint_every=20 training.rollout_every=20 \
 task.dataset.zarr_path="$EXP/data/adroit_door_medium.zarr" \
 task.critic_dataset.zarr_path="$EXP/data/adroit_door_medium.zarr" \
 task.finetune_dataset.zarr_path="$EXP/data/adroit_door_medium.zarr" \
 task.scale_dataset.zarr_path="$EXP/data/adroit_door_medium.zarr" \
 use_action_embed=False horizon=3 n_action_steps=1 n_obs_steps=3 ft_all_actions=False \
 num_inference_steps=10 policy.scheduler_type=ddim policy.model=skipnet policy.act=mish \
 feature_type=2D use_agent_pos=True policy.use_visual=True encoder_output_dim=64 \
 task.env_runner.eval_episodes=30 task.env_runner.env_num=2 ++task.env_runner.with_pointcloud=False \
 policy.img_shape='[3,224,224]' policy.use_aug=True \
 task.dataset.pre_image_norm=True task.critic_dataset.pre_image_norm=True \
 task.finetune_dataset.pre_image_norm=True task.scale_dataset.pre_image_norm=True \
 use_recon=True use_vib=True dynamics_type=diffusion training.num_epochs=600 \
 training.num_critic_epochs=600 dynamics.dynamics_max_epochs=350 \
 dataloader.batch_size=128 val_dataloader.batch_size=128 dataloader.num_workers=4 val_dataloader.num_workers=2 \
 optimizer.lr=2e-4 critic.q_lr=2e-4 critic.v_lr=2e-4 dynamics.dynamics_lr=4.4e-4 \
 encoder_type=resnet encoders.resnet.rgb_model.weights=r3m encoders.resnet.share_rgb_model=False \
 encoders.resnet.recon_loss_weight=0.05 encoders.resnet.kl_beta=1e-4 kl_annealing=False \
 offline=True online=True distill_phase=null unio4.bppo_steps=6000 unio4.bppo_lr=5.66e-6 unio4.rollout_length=10 \
 clip_std_max=0.8 ppo.max_train_steps=1000000 ppo.lr_a=4.6e-6 ppo.K_epochs=5 \
 ++ppo.use_vec_env_online=True ++ppo.train_env_num=4 ++ppo.eval_env_num=2 \
 ppo.save_online_cp=True ppo.online_cp_save_freq=10 "$@"
