from pathlib import Path
import argparse, json, numpy as np, torch, time
from paired_env import PairedPegEnv
from metaworld.policies import SawyerPegInsertionSideV2Policy

p=argparse.ArgumentParser();p.add_argument('--shard',type=int,default=0);p.add_argument('--count',type=int,default=34);p.add_argument('--smoke',action='store_true');p.add_argument('--gain',type=float,default=25);a=p.parse_args()
root=Path(__file__).parent; dest=root/'data'/f'seeded_gain{int(a.gain)}_shard{a.shard}';dest.mkdir(parents=True,exist_ok=True)
torch.manual_seed(4100+a.shard);np.random.seed(4100+a.shard)
env=PairedPegEnv(device='cuda:0',num_points=512,rgb_size=84,use_point_crop=True)
expert=SawyerPegInsertionSideV2Policy();accepted=0;attempts=[];started=time.time()
for attempt in range(a.count*5):
    seed=10000+a.shard*10000+attempt
    env.seed(seed);obs=env.reset();rows=[];succ=[]
    initial=dict(qpos=env.env.sim.data.qpos.copy(),qvel=env.env.sim.data.qvel.copy(),target=env.env._target_pos.copy(),rand_vec=env.env._last_rand_vec.copy())
    for t in range(200):
        action=expert.get_action(obs['full_state']);action[:3]*=a.gain/25
        action=np.clip(action,-1,1).astype(np.float32)
        nxt,r,done,info=env.step(action);success=bool(info['success']);succ.append(success)
        row=dict(img=obs['image'].transpose(1,2,0),next_img=nxt['image'].transpose(1,2,0),depth=obs['depth'],next_depth=nxt['depth'],point_cloud=obs['point_cloud'],next_point_cloud=nxt['point_cloud'],state=obs['agent_pos'],next_state=nxt['agent_pos'],full_state=obs['full_state'],next_full_state=nxt['full_state'],action=action,reward=np.array([r],np.float32),done=np.array([done and not info['TimeLimit.truncated']],bool),timeout=np.array([info['TimeLimit.truncated']],bool),success=np.array([success],bool))
        rows.append(row);obs=nxt
        if done:break
    passed=any(succ);attempts.append(dict(seed=seed,success=passed,success_steps=sum(succ),steps=len(rows)))
    if passed:
        data={k:np.stack([r[k] for r in rows]) for k in rows[0]}
        last=expert.get_action(obs['full_state']);last[:3]*=a.gain/25
        last=np.clip(last,-1,1).astype(np.float32)
        data['next_action']=np.concatenate([data['action'][1:],last[None]])
        ret=np.zeros_like(data['reward']);g=0.
        for i in reversed(range(len(rows))):g=float(data['reward'][i,0])+.99*g;ret[i,0]=g
        data['return']=ret
        out=dest/f'episode_{accepted:03d}.npz'
        assert not out.exists(),out
        np.savez_compressed(out,**data,seed=seed,**{'initial_'+k:v for k,v in initial.items()})
        accepted+=1
    (dest/'status.json').write_text(json.dumps(dict(accepted=accepted,attempted=len(attempts),attempts=attempts,elapsed=time.time()-started),indent=2))
    print(json.dumps(dict(shard=a.shard,accepted=accepted,attempted=len(attempts),seed=seed,success=passed,success_steps=sum(succ),elapsed=time.time()-started)),flush=True)
    if a.smoke or accepted>=a.count:break
assert accepted>0,'Expert produced no successful trajectories'
