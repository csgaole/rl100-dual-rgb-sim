from pathlib import Path
import os,json,time,subprocess,signal,torch,dill,re
r=Path(__file__).parent;state=json.loads((r/'cap400_jobs.json').read_text());jobs=state['jobs'];active={};tasks={};seen={}
def live(pid):
 try:return Path(f'/proc/{pid}/stat').read_text().split()[2]!='Z'
 except FileNotFoundError:return False
def save():
 state.update(updated_unix=time.time(),evaluations=list(tasks.values()))
 p=r/'cap400_status.tmp';p.write_text(json.dumps(state,indent=2));p.replace(r/'cap400_status.json')
 # Keep the standard status endpoint current for later status queries.
 p=r/'pipeline_status.tmp';p.write_text(json.dumps(dict(updated_unix=time.time(),jobs=jobs,epoch_cap=400,evaluation_status_file=str(r/'cap400_status.json'),next_stage='Review matched checkpoint validation before offline RL'),indent=2));p.replace(r/'pipeline_status.json')
while True:
 for j in jobs:
  run=Path(j['run_dir']);dest=Path(j['retained_dir'])
  if j['phase']=='training_to_400':
   cp=run/'checkpoints/latest.ckpt';signature=cp.stat().st_mtime_ns
   if seen.get(str(cp))!=signature and time.time()-cp.stat().st_mtime>1:
    try:
     p=torch.load(cp,map_location='cpu',pickle_module=dill);ep=int(dill.loads(p['pickles']['epoch']))
     target=dest/f'epoch_{ep:04d}_latest.ckpt'
     if not target.exists():torch.save(p,target,pickle_module=dill)
     seen[str(cp)]=signature
     if ep>=400:
      assert j['run_dir'].encode() in Path(f"/proc/{j['pid']}/cmdline").read_bytes()
      os.killpg(j['pid'],signal.SIGTERM);j['phase']='stopped_at_user_cap';j['last_saved_epoch']=ep
    except (EOFError,RuntimeError,OSError) as e:print('CHECKPOINT_RETRY',str(e),flush=True)
   if j['phase']=='training_to_400' and not live(j['pid']):j['phase']='training_ended_before_cap'
  for cp in sorted(dest.glob('*.ckpt')):
   ep=int(re.search(r'epoch_(\d+)',cp.name)[1]);key=f"{j['name']}_epoch{ep}"
   if key not in tasks:
    tasks[key]=dict(key=key,run_dir=str(run),checkpoint=str(cp),epoch=ep,gpu=1 if j['name']=='rgb_bc_seed100' else j['gpu'],phase='queued',result=str(run/f'cap400_eval/epoch_{ep:04d}.json'))
 for gpu,(p,key) in list(active.items()):
  code=p.poll()
  if code is None:continue
  t=tasks[key];t['phase']='complete' if code==0 and Path(t['result']).exists() else 'failed';t['exit_code']=code
  if t['phase']=='complete':t['metrics']=json.loads(Path(t['result']).read_text())
  del active[gpu]
 for t in tasks.values():
  gpu=t['gpu']
  if t['phase']!='queued' or gpu in active:continue
  usage=int(subprocess.check_output(['nvidia-smi',f'--id={gpu}','--query-gpu=memory.used','--format=csv,noheader,nounits'],text=True).strip())
  if usage>2000:continue
  log=r/f"logs/cap400_{t['key']}.log"
  p=subprocess.Popen(['bash',str(r/'run.sh'),str(r/'eval_checkpoint.py'),t['run_dir'],t['checkpoint'],t['result']],env=dict(os.environ,GPU_ID=str(gpu)),stdout=open(log,'x'),stderr=subprocess.STDOUT,stdin=subprocess.DEVNULL,start_new_session=True)
  t.update(phase='running',pid=p.pid,log=str(log));active[gpu]=(p,t['key'])
 save()
 if not any(j['phase']=='training_to_400' for j in jobs) and all(t['phase'] in ['complete','failed'] for t in tasks.values()):break
 time.sleep(2)
print('CAP400_AND_COMPARISONS_COMPLETE',flush=True)
