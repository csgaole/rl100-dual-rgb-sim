from pathlib import Path
import sys, os, json, subprocess, hashlib, copy, time
import torch, dill
from omegaconf import OmegaConf
root=Path(__file__).resolve().parent
base=root.parent/'peg_rl_complete_v2/RL-100'
mode=sys.argv[1];assert mode in ('smoke','full','replicates')
manifest=root/f'rl_{mode}_jobs.json';assert not manifest.exists()
jobs=[]
assignments={'smoke':[(100,7,100)],'full':[(100,5,100),(300,6,300)],'replicates':[(200,2,100),(400,7,300)]}
for seed,gpu,bc_seed in assignments[mode]:
 source=root/f'runs/dual_rgb_bn_repaired_seed{bc_seed}'
 cp=list((source/'checkpoints').glob('epoch=*-test_mean_score=*.ckpt'));assert len(cp)==1
 cp=cp[0]
 if mode!='smoke':
  evaluation=json.loads((root/f'bc_selected_eval_seed{bc_seed}.json').read_text())
  assert evaluation['sha256']==hashlib.sha256(cp.read_bytes()).hexdigest()
  assert evaluation['episodes']==100 and evaluation['success_rate']>0
  assert json.loads((root/'rl_smoke_seed100/pipeline_status.json').read_text())['phase']=='complete'
 used=int(subprocess.check_output(['nvidia-smi',f'--id={gpu}','--query-gpu=memory.used','--format=csv,noheader,nounits'],text=True));assert used<1000,(gpu,used)
 run=root/f'rl_{mode}_seed{seed}';assert not run.exists()
 (run/'checkpoints').mkdir(parents=True)
 cfg=OmegaConf.load(source/'.hydra/config.yaml');OmegaConf.set_struct(cfg,False)
 cfg.training.seed=seed
 cfg.only_bc=False;cfg.offline=True;cfg.online=True;cfg.stage1_only=False;cfg.load_bc=False
 for role in ('dataset','critic_dataset','scale_dataset','finetune_dataset'):
  cfg.task[role]._target_='dual_rl_dataset.Dataset'
 cfg.training.resume=True;cfg.unio4.stage1_resume_dir=str(run)
 cfg.training.num_critic_epochs=1 if mode=='smoke' else 100
 cfg.dynamics.dynamics_max_epochs=1 if mode=='smoke' else 100
 cfg.dynamics.max_epochs_since_update=10
 cfg.unio4.bppo_steps=2 if mode=='smoke' else 2000
 cfg.unio4.eval_freq=2 if mode=='smoke' else 500
 cfg.unio4.eval_step=2 if mode=='smoke' else 50
 cfg.unio4.finetune_batch_size=32 if mode=='smoke' else 128
 cfg.ppo.max_train_steps=512 if mode=='smoke' else 1000000
 cfg.ppo.batch_size=128 if mode=='smoke' else 512
 cfg.ppo.mini_batch_size=32 if mode=='smoke' else 128
 cfg.ppo.K_epochs=2 if mode=='smoke' else 3
 cfg.ppo.evaluate_freq=512 if mode=='smoke' else 20000
 cfg.ppo.online_cp_save_freq=1 if mode=='smoke' else 10
 cfg.task.env_runner.eval_episodes=2 if mode=='smoke' else 20
 cfg.ema._target_='ema_repaired.EMAModel'
 cfg.exp_name=f'dual_rgb_rl_{mode}_{seed}'
 p=torch.load(cp,map_location='cpu',pickle_module=dill)
 p['state_dicts']['model']=copy.deepcopy(p['state_dicts']['ema_model'])
 p['pickles']['_output_dir']=dill.dumps(str(run));p['cfg']=cfg
 torch.save(p,run/'checkpoints/latest.ckpt',pickle_module=dill)
 name=f'dual_rgb_rl_{mode}_{seed}';OmegaConf.save(cfg,base/f'rl_100/config/{name}.yaml')
 (run/'bc_source.json').write_text(json.dumps(dict(checkpoint=str(cp),sha256=hashlib.sha256(cp.read_bytes()).hexdigest(),policy='ema_model',selection='validation top-k'),indent=2))
 log=root/f'logs/{name}.log'
 proc=subprocess.Popen(['bash',str(root/'run.sh'),str(root/'rl_pipeline.py'),name,str(run),mode],env=dict(os.environ,GPU_ID=str(gpu)),stdout=log.open('x'),stderr=subprocess.STDOUT,stdin=subprocess.DEVNULL,start_new_session=True)
 jobs.append(dict(seed=seed,bc_seed=bc_seed,gpu=gpu,pid=proc.pid,run_dir=str(run),log=str(log),started_unix=time.time(),offline_updates=cfg.unio4.bppo_steps,online_transitions=cfg.ppo.max_train_steps))
 manifest.write_text(json.dumps(jobs,indent=2));print(json.dumps(jobs[-1]),flush=True)
